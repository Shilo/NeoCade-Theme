    ```gdscript
    @tool
    extends EditorScript

    ## Phase 4 build-time helper. Generates Inter-Variable.tres + 5 FontVariation .tres
    ## using ResourceSaver.save() — Godot-native serialization, NOT hand-authored.
    ## DELETE this file in Phase 11 before distribution (alongside _phase4_verify.gd).
    ## Run via Godot Editor → File → Run, or: godot --headless --editor --script addons/neocade_theme/_phase4_import.gd

    func _run() -> void:
        var fonts_dir := "res://addons/neocade_theme/fonts/"

        # Step 1: ensure the TTF is imported. Calling load() on the TTF triggers Godot's
        # FontFile importer if .import sidecar is absent or stale; Godot writes the sidecar.
        var inter_ttf: FontFile = load(fonts_dir + "Inter-Variable.ttf") as FontFile
        assert(inter_ttf != null, "Inter-Variable.ttf failed to import — re-run after editor first-launch import pass")

        # Step 2: apply the GL-Compatibility-correct import settings. PITFALLS 5.5 + STACK.
        # We CANNOT override .import params from a runtime script directly (importer params
        # are file-side, not resource-side). Instead, we write a deterministic .import file
        # template that Godot will RE-READ on next import. The .import file format is a
        # Godot-native ini-style file; ResourceSaver does NOT manage it (it manages .tres).
        # Strategy: write the .import settings via Set-Content as the planner originally did
        # — BUT use the EXACT format Godot wrote in step 1 (read it back, mutate the [params]
        # block, write it back). This guarantees the format matches Godot's serializer.
        var import_path := fonts_dir + "Inter-Variable.ttf.import"
        var fa := FileAccess.open(import_path, FileAccess.READ)
        assert(fa != null, "Inter-Variable.ttf.import was not generated by the editor import pass")
        var raw := fa.get_as_text()
        fa.close()
        # Mutate the params block to enforce our required values (idempotent).
        var required := {
            "antialiasing": "1",          # Grayscale (PITFALLS 5.5)
            "hinting": "1",               # Light
            "subpixel_positioning": "2",  # Auto
            "generate_mipmaps": "true",
            "allow_system_fallback": "true",
            "multichannel_signed_distance_field": "false",
        }
        var mutated := raw
        for k in required.keys():
            var pattern := RegEx.new()
            pattern.compile("(?m)^%s=.*$" % k)
            var m := pattern.search(mutated)
            if m:
                mutated = pattern.sub(mutated, "%s=%s" % [k, required[k]])
            else:
                # Append under [params] if missing
                mutated = mutated.replace("[params]\n", "[params]\n%s=%s\n" % [k, required[k]])
        var fb := FileAccess.open(import_path, FileAccess.WRITE)
        fb.store_string(mutated)
        fb.close()

        # Step 3: author Inter-Variable.tres FontFile wrapper via ResourceSaver.
        # The wrapper exists so consumers have a stable preload-able resource path
        # (FontVariation.base_font expects a Font resource; we pass Inter-Variable.tres).
        # NOTE: Godot's FontFile is itself a Resource — we can save the loaded TTF as
        # a .tres directly (acts as a FontFile alias resource).
        var inter_tres_path := fonts_dir + "Inter-Variable.tres"
        var save_ok := ResourceSaver.save(inter_ttf, inter_tres_path)
        assert(save_ok == OK, "Failed to save Inter-Variable.tres: %d" % save_ok)

        # Step 4: author 5 FontVariation .tres files via ResourceSaver.
        # Each uses Inter-Variable.tres as base_font + sets variation_opentype.
        var inter_base: FontFile = load(inter_tres_path) as FontFile
        var variants := [
            {"file": "Inter-HeaderLarge.tres",  "wght": 800, "opsz": 32},
            {"file": "Inter-HeaderMedium.tres", "wght": 700, "opsz": 32},
            {"file": "Inter-HeaderSmall.tres",  "wght": 600, "opsz": 24},
            {"file": "Inter-Body.tres",         "wght": 400, "opsz": -1},  # opsz omitted
            {"file": "Inter-Caption.tres",      "wght": 400, "opsz": -1},
        ]
        for v in variants:
            var fv := FontVariation.new()
            fv.base_font = inter_base
            var open_type := {"wght": v.wght}
            if v.opsz > 0:
                open_type["opsz"] = v.opsz
            fv.variation_opentype = open_type
            var ok := ResourceSaver.save(fv, fonts_dir + v.file)
            assert(ok == OK, "Failed to save %s: %d" % [v.file, ok])

        print("✓ Phase 4 font materialization: TTF imported, .import params enforced, ",
              "Inter-Variable.tres + 5 FontVariation .tres files saved via ResourceSaver.")
    ```
